cmake_minimum_required(VERSION 3.21)
project(MelodayProjectM LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(GLEW REQUIRED)
add_library(Meloday.ProjectM SHARED bridge.cpp)
target_include_directories(Meloday.ProjectM PRIVATE "${PROJECTM_SOURCE}/src/api/include" "${PROJECTM_BUILD}/src/api/include")
target_link_libraries(Meloday.ProjectM PRIVATE "${PROJECTM_BUILD}/src/libprojectM/Release/projectM-4.lib" GLEW::glew opengl32)
set_property(TARGET Meloday.ProjectM PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
add_executable(projectm-smoke smoke.cpp)
target_link_libraries(projectm-smoke PRIVATE Meloday.ProjectM)
set_property(TARGET projectm-smoke PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
